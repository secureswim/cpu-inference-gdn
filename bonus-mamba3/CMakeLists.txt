cmake_minimum_required(VERSION 3.20)
project(mamba_bonus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Use GCC on x86-64 with AVX2 and FMA.")
endif()
add_library(mamba_cpu SHARED src/mamba.cpp)
set_target_properties(mamba_cpu PROPERTIES OUTPUT_NAME mamba3_decode)
target_compile_options(mamba_cpu PRIVATE -O3 -mavx2 -mfma)
option(BONUS_GGML "Build the optional ggml integration check" OFF)
if(BONUS_GGML)
  set(GGML_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/vendor/ggml" CACHE PATH "Prepared ggml tree")
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
  if(WIN32)
    add_compile_definitions(_WIN32_WINNT=0x0A00)
  endif()
  add_subdirectory("${GGML_SOURCE}" ggml EXCLUDE_FROM_ALL)
  add_executable(mamba_ggml src/mamba_ggml.cpp)
  target_link_libraries(mamba_ggml PRIVATE ggml ggml-cpu mamba_cpu)
endif()
