cmake_minimum_required(VERSION 3.20)
project(cpu_inference_project LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Explicit reproducible SIMD" FORCE)
set(GGML_AVX ON CACHE BOOL "AVX" FORCE)
set(GGML_AVX2 ON CACHE BOOL "AVX2" FORCE)
set(GGML_FMA ON CACHE BOOL "FMA" FORCE)
set(GGML_F16C ON CACHE BOOL "F16C" FORCE)
set(GGML_CUDA OFF CACHE BOOL "CPU only" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "CPU only" FORCE)
set(GGML_METAL OFF CACHE BOOL "CPU only" FORCE)
set(GGML_OPENMP ON CACHE BOOL "OpenMP" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "Avoid external compiler cache" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "Tests supplied by project" FORCE)
if(WIN32)
  add_compile_definitions(_WIN32_WINNT=0x0A00)
endif()
set(GGML_SOURCE "${CMAKE_SOURCE_DIR}/build/vendor/ggml" CACHE PATH "Pinned ggml source")
add_subdirectory(${GGML_SOURCE} ggml)
find_package(OpenMP REQUIRED)
add_executable(gdn src/gdn.cpp)
target_link_libraries(gdn PRIVATE ggml ggml-cpu OpenMP::OpenMP_CXX)
add_library(mamba_cpu SHARED src/mamba.cpp)
target_compile_options(mamba_cpu PRIVATE -O3 -mavx2 -mfma)
add_executable(mamba_ggml src/mamba_ggml.cpp)
target_link_libraries(mamba_ggml PRIVATE ggml ggml-cpu mamba_cpu)
enable_testing()
add_test(NAME gdn_correctness COMMAND gdn check)
add_test(NAME mamba_ggml_correctness COMMAND mamba_ggml)
add_executable(machine src/machine.cpp)
target_compile_options(machine PRIVATE -O3 -mavx2)
if(WIN32)
  target_link_libraries(gdn PRIVATE psapi)
  target_link_libraries(machine PRIVATE psapi)
endif()
add_executable(stream vendor/stream.c)
target_compile_definitions(stream PRIVATE STREAM_ARRAY_SIZE=20000000 NTIMES=20)
target_compile_options(stream PRIVATE -O3 -mavx2 -mfma)
target_link_libraries(stream PRIVATE OpenMP::OpenMP_C)
