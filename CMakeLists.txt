cmake_minimum_required(VERSION 3.20)
# Gate configuration before compiler/source probing, including manual CMake use.
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/system_config.py" --for-build
  RESULT_VARIABLE SETUP_STATUS ERROR_VARIABLE SETUP_ERROR)
if(NOT SETUP_STATUS EQUAL 0)
  message(FATAL_ERROR "${SETUP_ERROR}")
endif()
project(cpu_inference_project LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_BINARY_DIR}/bin")
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(GGML_NATIVE ON CACHE BOOL "Build for the detected host CPU" FORCE)
foreach(BACKEND CUDA VULKAN METAL BLAS ACCELERATE)
  set(GGML_${BACKEND} OFF CACHE BOOL "CPU only, no external accelerators" FORCE)
endforeach()
set(GGML_OPENMP ON CACHE BOOL "OpenMP" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "Avoid external compiler cache" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "Tests supplied by project" FORCE)
if(WIN32)
  add_compile_definitions(_WIN32_WINNT=0x0A00 NOMINMAX)
endif()
set(GGML_SOURCE "${CMAKE_SOURCE_DIR}/build/vendor/ggml" CACHE PATH "Pinned ggml source")
add_subdirectory(${GGML_SOURCE} ggml)
find_package(OpenMP REQUIRED)
add_custom_target(check_system ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/system_config.py" --for-build
  VERBATIM)
add_executable(gdn src/gdn.cpp)
target_link_libraries(gdn PRIVATE ggml ggml-cpu OpenMP::OpenMP_CXX)
add_executable(machine src/machine.cpp)
add_executable(stream vendor/stream.c)
if(MSVC)
  target_include_directories(stream PRIVATE "${CMAKE_SOURCE_DIR}/src/compat/msvc")
endif()
add_executable(ceilings src/ceilings.cpp)
target_link_libraries(ceilings PRIVATE OpenMP::OpenMP_CXX)
target_link_libraries(stream PRIVATE OpenMP::OpenMP_C)
set(PROJECT_STREAM_ARRAY_SIZE 20000000 CACHE STRING "STREAM elements per array (adapted to LLC)")
target_compile_definitions(stream PRIVATE STREAM_ARRAY_SIZE=${PROJECT_STREAM_ARRAY_SIZE} NTIMES=20)
foreach(TARGET gdn machine stream ceilings)
  add_dependencies(${TARGET} check_system)
  if(MSVC)
    target_compile_options(${TARGET} PRIVATE /O2)
  else()
    target_compile_options(${TARGET} PRIVATE -O3)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
      target_compile_options(${TARGET} PRIVATE -mcpu=native)
    else()
      target_compile_options(${TARGET} PRIVATE -march=native)
    endif()
  endif()
endforeach()
if(WIN32)
  target_link_libraries(gdn PRIVATE psapi)
  target_link_libraries(machine PRIVATE psapi)
endif()
enable_testing()
add_test(NAME gdn_correctness COMMAND gdn check)
